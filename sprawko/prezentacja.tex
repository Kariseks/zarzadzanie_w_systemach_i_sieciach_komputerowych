\begin{frame}{Cel i zakres}
	\begin{itemize}
		\item Wykrywanie krawędzi operatorami gradientowymi (Sobel, Prewitt, Scharr, Roberts) na obrazach do 59~Mpix.
		\item Pytanie badawcze: ile czasu przetwarzania dają kolejne zmiany w kodzie i w sposobie zarządzania wątkami?
		\item Zbadane mechanizmy:
		\begin{itemize}
			\item optymalizacje jednowątkowe (V0--V8), w tym SIMD AVX2,
			\item pula wątków z harmonogramami static / cyclic / dynamic / guided,
			\item synchronizacja: bariera, redukcja maksimum, false sharing,
			\item przypinanie wątków do rdzeni, OpenMP, GPU (OpenCL).
		\end{itemize}
		\item Kryteria: czas zegarowy, przyspieszenie $S(N) = T(1)/T(N)$, efektywność $E(N) = S(N)/N$, poprawność względem wersji referencyjnej.
	\end{itemize}
\end{frame}

%----------------------------------------------------------------------
\begin{frame}{Sformułowanie problemu}
	\[
		G(x, y) = \sqrt{G_x^{2} + G_y^{2}}, \qquad G_x = K_x * I, \quad G_y = K_y * I,
		\qquad \text{wynik} = \operatorname{round}\!\left(255 \cdot \frac{G}{\max G}\right)
	\]
	\begin{itemize}
		\item Parametry: obraz $W \times H$, maska $k \times k$ ($k = 3, 5, 7$), liczba wątków $N$, harmonogram, rozmiar fragmentu.
		\item Złożoność: $O(W H k^{2})$ dla splotu 2D, $O(W H k)$ dla maski rozdzielnej (Sobel, Prewitt, Scharr).
		\item Każdy piksel zależy tylko od otoczenia -- zadanie łatwo zrównoleglane, ale:
		\begin{itemize}
			\item normalizacja do $\max G$ wymaga drugiego przebiegu, redukcji i bariery,
			\item wczytanie i zapis pliku są sekwencyjne (prawo Amdahla),
			\item splot $3 \times 3$ wykonuje mało obliczeń na bajt -- ogranicza go pamięć.
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Implementacja}
	\begin{columns}[T]
		\begin{column}{0.52\textwidth}
			\begin{itemize}
				\item C++17, CMake; biblioteka \kod{edge\_core} bez zależności.
				\item Jedna funkcja \kod{runVariant()} wywołuje każdą wersję -- te same granice pomiaru.
				\item \kod{edge\_bench} (Qt Core): eksperyment z pliku JSON, surowe wyniki do CSV, opis warunków do \kod{meta.json}.
				\item Python: pomiary po kolei (blokada, kontrola zajętości CPU), wykresy i tabele do sprawozdania.
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\begin{itemize}
				\item Poprawność: każdy wariant porównywany z V0 dla obrazów od $1 \times 1$ do $1921 \times 1081$, 6 masek, 3 norm, 3 skalowań i wielu konfiguracji wątków.
				\item Wszystkie warianty, także GPU, dają wynik \textbf{identyczny co do bajtu} z V0.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Warianty kodu -- każdy dodaje jedną zmianę}
	\footnotesize
	\begin{tabular}{@{}lp{0.76\textwidth}@{}}
		\toprule
		Wariant & Zmiana \\
		\midrule
		V0 & naiwny: \kod{std::clamp} przy każdym odczycie, maska z \kod{std::vector}, bufory alokowane w każdym wywołaniu \\
		V1 & brzegi obrazu obsłużone poza główną pętlą \\
		V2 & maska jako stała czasu kompilacji (szablon): rozwinięte pętle, bez mnożeń przez 0 \\
		V3 & splot rozdzielny: dwa sploty 1D, bufor $k$ wierszy ($k^2 \to 2k$ mnożeń) \\
		V4 & bufory \kod{int16}, wyrównane do 64~B, nie alokowane ponownie \\
		V5 & pętle przepisane pod automatyczną wektoryzację (AVX2) \\
		V6 & ręczne SIMD AVX2: 16 pikseli w przebiegu poziomym, 8 w pionowym \\
		V8 & V6 + przeliczenie zamiast bufora modułu (2\x{} obliczenia, bez 8~B/piksel ruchu w pamięci) \\
		\midrule
		T & pula wątków + scheduler, dowolne jądro V0--V8 \\
		OMP & \kod{\#pragma omp parallel for schedule(runtime)}, te same jądra \\
		GPU, GPUC, GPUL & OpenCL: kernel naiwny / maska wkompilowana / kafel w pamięci lokalnej \\
		\bottomrule
	\end{tabular}
\end{frame}

\begin{frame}{Wnioski}
	\small
	\begin{enumerate}
		\item Optymalizacje jednowątkowe dały \wynik{e1/V8/4k/przysp}\x{} (4K) -- więcej niż 24 wątki dla kodu naiwnego (\wynik{e3/V0/4k/Smax}\x).
		\item Im szybsze jądro, tym gorsze skalowanie: ograniczeniem staje się pamięć i część sekwencyjna (T/V8: maks.\ \wynik{e3/V8/4k/Smax}\x).
		\item Wczytanie i zapis JPG ($s = \wynik{e9/4k-jpg/s}\,\%$) przekreślają zysk z wątków dla całego programu.
		\item Sposób synchronizacji zmienia czas o rzędy wielkości: false sharing \wynik{e8b/px-nopad/t24}~ms, mutex \wynik{e8b/mutex/t24}~ms wobec \wynik{e8b/local/t24}~ms.
		\item Przy stałym koszcie wiersza najlepszy jest prosty podział statyczny.
		\item GPU klasy GTX 1050 Ti przegrywa z 12-rdzeniowym CPU z AVX2 -- przesył i słaba moc obliczeniowa.
	\end{enumerate}
	\vspace{0.5em}
